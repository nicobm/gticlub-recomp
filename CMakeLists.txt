cmake_minimum_required(VERSION 3.24)
project(GTiPort LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(SDL3 REQUIRED CONFIG)
find_package(ZLIB REQUIRED)

add_library(gti_formats STATIC
  src/formats/ep3.cpp
  src/formats/edge_indexes.cpp
  src/formats/gcm_texture.cpp
)
target_include_directories(gti_formats PUBLIC src)
target_link_libraries(gti_formats PUBLIC ZLIB::ZLIB)

add_executable(gtiport WIN32
  src/main.cpp
  src/render/gl_loader.cpp
  src/render/renderer.cpp
  src/game/collision.cpp
  src/game/vehicle.cpp
  src/audio/vag.cpp
  src/audio/engine_audio.cpp
)
target_link_libraries(gtiport PRIVATE gti_formats SDL3::SDL3-static opengl32)

# SPU interpreter test bench
add_executable(spu_test src/tools/spu_test.cpp src/recomp/spu.cpp)
target_include_directories(spu_test PRIVATE src)

# headless physics test bench
add_executable(physlab src/tools/physlab.cpp src/game/collision.cpp src/game/vehicle.cpp)
target_link_libraries(physlab PRIVATE gti_formats)

# CLI tool to dump/validate packages (no window)
add_executable(ep3tool src/tools/ep3tool.cpp)
target_link_libraries(ep3tool PRIVATE gti_formats)

if(MINGW)
  # self-contained exe: no libc++/libunwind/winpthread DLL dependencies
  foreach(t gtiport ep3tool physlab)
    target_link_options(${t} PRIVATE -static)
  endforeach()
endif()

# statically recompiled original PPU code (generated locally by tools/ppc2cpp.py from the user's own EBOOT)
if(EXISTS ${CMAKE_SOURCE_DIR}/src/recomp/gen/physics.cpp)
  add_executable(recomp_test src/tools/recomp_test.cpp src/recomp/runtime.cpp src/recomp/guest.cpp src/recomp/gen/physics.cpp)
  target_link_libraries(recomp_test PRIVATE gti_formats)
  target_compile_options(recomp_test PRIVATE -Wno-unused-label -Wno-unused-variable -ffp-contract=off)
  if(MINGW)
    target_link_options(recomp_test PRIVATE -static)
  endif()
endif()

# whole game recompiled (generated locally by tools/ppc2cpp.py, SPLIT=48) — large; built only on demand
file(GLOB RECOMP_FULL_SRC CONFIGURE_DEPENDS src/recomp/full/*.cpp)
if(RECOMP_FULL_SRC)
  add_library(recomp_full STATIC ${RECOMP_FULL_SRC} src/recomp/runtime.cpp src/recomp/hle.cpp)
  target_include_directories(recomp_full PUBLIC src)
  # -O2 and FMA: the game code is half vector maths (vmaddfp: std::fmaf was a library call per lane at -O1)
  target_compile_options(recomp_full PRIVATE -O2 -mavx2 -mfma -Wno-unused-label -Wno-unused-variable -ffp-contract=off -w)
  # the port drives with the original physics when started with --orig <snapshot>
  target_sources(gtiport PRIVATE src/game/orig_physics.cpp src/recomp/guest.cpp)
  target_link_libraries(gtiport PRIVATE recomp_full)
  target_compile_definitions(gtiport PRIVATE GTI_ORIG_PHYSICS)
  add_executable(recomp_frame src/tools/recomp_frame.cpp src/recomp/guest.cpp)
  target_link_libraries(recomp_frame PRIVATE recomp_full gti_formats)
  if(MINGW)
    target_link_options(recomp_frame PRIVATE -static)
  endif()
endif()
file(GLOB SPU_GEN_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/recomp/spu_gen/*.cpp)
if(SPU_GEN_SRC)
  # generated by tools/spu2cpp.py; helpers (spu_ops.h) use SSSE3/SSE4.1, no FMA contraction to stay bit-exact
  set_source_files_properties(${SPU_GEN_SRC} PROPERTIES COMPILE_OPTIONS "-mavx2;-fno-strict-aliasing;-ffp-contract=off;-w;-gline-tables-only")
  # recompiled SPU code against the interpreter, instruction by instruction
  add_executable(spu_diff src/tools/spu_diff.cpp src/recomp/spu.cpp ${SPU_GEN_SRC})
  set_source_files_properties(src/recomp/edge_cull.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-gline-tables-only")
  add_executable(cull_replay src/tools/cull_replay.cpp src/recomp/spu.cpp src/recomp/edge_cull.cpp ${SPU_GEN_SRC})
  target_include_directories(cull_replay PRIVATE src)
  if(MINGW)
    target_link_options(cull_replay PRIVATE -static)
  endif()
  set_source_files_properties(src/recomp/edge_native.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-gline-tables-only")
  add_executable(fn_replay src/tools/fn_replay.cpp src/recomp/spu.cpp src/recomp/edge_native.cpp ${SPU_GEN_SRC})
  target_include_directories(fn_replay PRIVATE src)
  if(MINGW)
    target_link_options(fn_replay PRIVATE -static)
  endif()
  target_include_directories(spu_diff PRIVATE src)
  if(MINGW)
    target_link_options(spu_diff PRIVATE -static)
  endif()
endif()
# Vulkan back end: the official headers (third_party/vulkan; the functions are loaded at run time through SDL) and
# glslang (third_party/glslang, Khronos) to turn the translated GLSL into SPIR-V inside the game
if(RECOMP_FULL_SRC AND EXISTS ${CMAKE_SOURCE_DIR}/third_party/glslang/CMakeLists.txt)
  set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
  set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
  set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
  set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
  set(ENABLE_PCH OFF CACHE BOOL "" FORCE)
  set(BUILD_EXTERNAL OFF CACHE BOOL "" FORCE)
  set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
  set(GLSLANG_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/glslang EXCLUDE_FROM_ALL)
endif()
if(RECOMP_FULL_SRC)
  # whole game booted natively from its entry point (system layer: src/recomp/lv2.cpp)
  add_executable(recomp_boot src/tools/recomp_boot.cpp src/recomp/guest.cpp src/recomp/lv2.cpp src/recomp/netplay.cpp src/recomp/spu.cpp src/recomp/rsx_core.cpp src/recomp/rsx_gl.cpp src/recomp/rsx_d3d11.cpp src/recomp/rsx_vulkan.cpp src/recomp/ripprof.cpp src/recomp/host_win.cpp src/recomp/button_glyphs.cpp src/recomp/settings.cpp src/recomp/audio_out.cpp src/recomp/atrac3.cpp src/recomp/host_camera.cpp src/recomp/edge_cull.cpp src/recomp/edge_native.cpp ${SPU_GEN_SRC})
  target_link_libraries(recomp_boot PRIVATE recomp_full gti_formats SDL3::SDL3-static opengl32)
  target_include_directories(recomp_boot PRIVATE third_party/vulkan/include third_party/glslang)
  target_link_libraries(recomp_boot PRIVATE glslang SPIRV glslang-default-resource-limits)
  # the game's icon (assets/racing_club.ico is the user's own artwork and is not in the repository) and a copy of the
  # executable under the game's name: started with a double click it finds the game files by itself
  if(WIN32 AND EXISTS ${CMAKE_SOURCE_DIR}/assets/racing_club.ico)
    enable_language(RC)
    target_sources(recomp_boot PRIVATE src/tools/recomp_boot.rc)
    set_source_files_properties(src/tools/recomp_boot.rc PROPERTIES OBJECT_DEPENDS ${CMAKE_SOURCE_DIR}/assets/racing_club.ico)
  endif()
  if(WIN32)
    add_custom_command(TARGET recomp_boot POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:recomp_boot> "$<TARGET_FILE_DIR:recomp_boot>/Racing Club+.exe" VERBATIM)
  endif()
  # line tables (no effect on speed) so the RIPPROF samples of the system layer resolve to source lines
  set_source_files_properties(src/recomp/lv2.cpp src/recomp/rsx_core.cpp src/recomp/rsx_gl.cpp src/recomp/rsx_d3d11.cpp src/recomp/rsx_vulkan.cpp PROPERTIES COMPILE_OPTIONS "-gline-tables-only")
  if(WIN32)
    target_link_libraries(recomp_boot PRIVATE dwmapi gdiplus shlwapi d3d11 dxgi ws2_32 ole32 oleaut32)
  endif()
  if(MINGW)
    target_link_options(recomp_boot PRIVATE -static)
  endif()
  add_executable(recomp_fmt src/tools/recomp_fmt.cpp src/recomp/guest.cpp)
  target_link_libraries(recomp_fmt PRIVATE recomp_full gti_formats)
  add_executable(recomp_probe src/tools/recomp_probe.cpp src/recomp/guest.cpp)
  target_link_libraries(recomp_probe PRIVATE recomp_full gti_formats)
  if(MINGW)
    target_link_options(recomp_probe PRIVATE -static)
  endif()
endif()

# "Instalar Racing Club+.exe": opens the graphical installer (instalador.pyw) without a console window
if(WIN32)
  add_executable(instalador WIN32 src/tools/instalador_launcher.c)
  set_target_properties(instalador PROPERTIES OUTPUT_NAME "Instalar Racing Club+")
  target_link_options(instalador PRIVATE -municode -static -s)
  if(EXISTS ${CMAKE_SOURCE_DIR}/assets/racing_club.ico)
    target_sources(instalador PRIVATE src/tools/instalador_launcher.rc)
  endif()
endif()

target_include_directories(atrac3_test PRIVATE src)
target_compile_options(atrac3_test PRIVATE -O2 -mavx2 -mfma)
if(MINGW)
  target_link_options(atrac3_test PRIVATE -static)
